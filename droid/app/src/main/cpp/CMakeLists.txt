cmake_minimum_required(VERSION 3.22)
project(Ukuvota C)
foreach(required UKU_GENERATED UKU_RAYLIB UKU_OQS UKU_SQLITE UKU_ZIRAN)
    if(NOT DEFINED ${required})
        message(FATAL_ERROR "${required} must be resolved by scripts/android-build.sh")
    endif()
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PLATFORM Android CACHE STRING "" FORCE)
set(GRAPHICS GRAPHICS_API_OPENGL_ES2 CACHE STRING "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
add_subdirectory("${UKU_RAYLIB}" "${CMAKE_BINARY_DIR}/raylib")
set(OQS_BUILD_ONLY_LIB ON CACHE BOOL "" FORCE)
set(OQS_USE_OPENSSL OFF CACHE BOOL "" FORCE)
set(OQS_DIST_BUILD OFF CACHE BOOL "" FORCE)
set(OQS_OPT_TARGET generic CACHE STRING "" FORCE)
set(OQS_MINIMAL_BUILD SIG_ml_dsa_44 CACHE STRING "" FORCE)
add_subdirectory("${UKU_OQS}" "${CMAKE_BINARY_DIR}/oqs")
file(GLOB APP_SOURCES CONFIGURE_DEPENDS "${UKU_GENERATED}/*.c")
add_library(main SHARED ${APP_SOURCES} android_bridge.c "${UKU_SQLITE}/sqlite3.c")
target_include_directories(main PRIVATE "${UKU_GENERATED}" "${UKU_ZIRAN}/include"
    "${ANDROID_NDK}/sources/android/native_app_glue")
target_compile_definitions(main PRIVATE ANDROID_BUILD=1 PLATFORM_ANDROID
    SQLITE_OMIT_LOAD_EXTENSION _DEFAULT_SOURCE)
target_compile_options(main PRIVATE -Os -ffunction-sections -fdata-sections)
target_link_options(main PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
target_link_libraries(main raylib oqs android log m)
